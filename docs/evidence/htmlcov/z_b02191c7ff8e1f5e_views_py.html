14def get_filtered_components(request): 

15 """ 

16 Helper function to filter active components by category and availability status. 

17 Used by both report page view and CSV export view. 

18 """ 

19 components = Component.objects.filter(is_active=True) 

20 category = request.GET.get("category", "").strip() 

21 availability = request.GET.get("availability", "").strip() 

22 

23 # FR-11: Filter by category 

24 if category: 

25 components = components.filter(category=category) 

26 

27 # FR-11: Filter by availability status 

28 if availability == "In Stock": 

29 components = components.filter(quantity__gt=F("minimum_stock")) 

30 elif availability == "Low Stock": 

31 components = components.filter(quantity__gt=0, quantity__lte=F("minimum_stock")) 

32 elif availability == "Out of Stock": 

33 components = components.filter(quantity__lte=0) 

34 

35 return components.order_by("component_id"), category, availability

41def dashboard(request): 

42 """Show live inventory totals and components at or below their stock threshold.""" 

43 active_components = Component.objects.filter(is_active=True) 

44 low_stock_components = active_components.filter(quantity__lte=F("minimum_stock")) 

45 available_stock = active_components.aggregate(total=Sum("quantity"))["total"] or 0 

46 return render(request, "reports/dashboard.html", { 

47 "page_title": "Inventory Dashboard", 

48 "total_components": active_components.count(), 

49 "available_stock": available_stock, 

50 "low_stock_count": low_stock_components.count(), 

51 "low_stock_components": low_stock_components.order_by("component_id"), 

52 })

57def report_list(request): 

58 """ 

59 Generate and filter inventory reports (FR-11: ESIM-F-011 / TC-REP-01). 

60 Displays component details, stock quantities, and availability statuses with filtering options. 

61 """ 

62 if request.GET.get("format") == "csv" or request.GET.get("export") == "csv": 

63 return export_report_csv(request) 

64 

65 components, selected_category, selected_availability = get_filtered_components(request) 

66 

67 # Calculate summary metrics for the report 

68 total_items = components.count() 

69 total_units = sum(c.quantity for c in components) 

70 total_value = sum(c.quantity * c.price for c in components) 

71 in_stock_count = sum(1 for c in components if c.availability == "In Stock") 

72 low_stock_count = sum(1 for c in components if c.availability == "Low Stock") 

73 out_of_stock_count = sum(1 for c in components if c.availability == "Out of Stock") 

74 

75 context = { 

76 "page_title": "Inventory Reports", 

77 "components": components, 

78 "categories": Component.CATEGORY_CHOICES, 

79 "selected_category": selected_category, 

80 "selected_availability": selected_availability, 

81 "availability_choices": ["In Stock", "Low Stock", "Out of Stock"], 

82 "total_items": total_items, 

83 "total_units": total_units, 

84 "total_value": total_value, 

85 "in_stock_count": in_stock_count, 

86 "low_stock_count": low_stock_count, 

87 "out_of_stock_count": out_of_stock_count, 

88 "is_filtered": bool(selected_category or selected_availability), 

89 } 

90 return render(request, "reports/report_list.html", context)
